function lines(c, text, maxWidth) {
  const result = [];
  for (const paragraph of text.split('\n')) {
    let line = '';
    for (const char of Array.from(paragraph)) {
      if (line && c.measureText(line + char).width > maxWidth) { result.push(line); line = char; }
      else line += char;
    }
    result.push(line);
  }
  return result;
}
function textBox(c, text, x, y, width, height, size, color = INK, family = sans, center = false) {
  if (!text.trim()) return;

  do { c.font = `${size}px ${family}`; wrapped = lines(c, text, width); lineHeight = size * 1.5; if (wrapped.length * lineHeight <= height) break; size -= 1; } while (size > 8);

}

function frameAnimation(duration,callback,signal) {
  return new Promise((resolve,reject)=>{
    const start=performance.now();
    function frame(now) {
      if(signal?.aborted) {reject(new DOMException('已取消','AbortError'));return;}
      const p=Math.min(1,(now-start)/Math.max(duration,1));callback(p);
      if(p<1) requestAnimationFrame(frame);else resolve();
    }
    requestAnimationFrame(frame);
  });
}

}

    scene(c,out.width,out.height,0);
    stream=out.captureStream(30);
    recorder=new MediaRecorder(stream,{mimeType:format[0],videoBitsPerSecond:8000000});
    recorder.ondataavailable=event=>{if(event.data.size)chunks.push(event.data);};
    stopPromise=new Promise(resolve=>{recorder.onstop=resolve;});
    recorder.onerror=()=>{recordingError=new Error('浏览器未能完成视频编码，请换用 WebM 格式重试。');cancellation.abort();};
    recorder.start(250);
    const hold=Number($('#hold').value)*1000,total=spreads.length*hold+(spreads.length-1)*TURN;
    let elapsed=0;
    function render(index,progress){scene(c,out.width,out.height,index,progress);draw(index,progress);}
    for(let index=0;index<spreads.length;index++){
      state=index;update();
      await frameAnimation(hold,p=>{render(index,0);$('#progress').value=(elapsed+p*hold)/total;},signal);elapsed+=hold;
      if(index<spreads.length-1){
        await frameAnimation(TURN,p=>{render(index,ease(p));$('#progress').value=(elapsed+p*TURN)/total;},signal);elapsed+=TURN;
      }
    }
    recorder.stop();await stopPromise;
    if(!chunks.length)throw new Error('没有录制到画面，请重新导出。');
    const blob=new Blob(chunks,{type:format[0]});
    if(lastVideoURL)URL.revokeObjectURL(lastVideoURL);
    lastVideoURL=URL.createObjectURL(blob);
    const link=document.createElement('a');link.href=lastVideoURL;link.download=filename(format[2]);link.textContent='再次下载翻书动画';
    $('#status').replaceChildren(document.createTextNode('动画已生成。'),link);
    link.click();$('#progress').value=1;
  } catch(error) {
    $('#status').textContent=recordingError?.message||(error.name==='AbortError'?'已取消导出，书本和已有文件都保留。':`导出失败：${error.message}`);
  } finally {
    if(recorder&&recorder.state!=='inactive'){recorder.stop();await stopPromise;}
    if(stream)stream.getTracks().forEach(track=>track.stop());
    cancellation=null;recording=false;state=oldState;stopButton.hidden=true;$('#progress').hidden=true;draw();update();
  }
}
